% ── A. The hash chain, precisely ──
\masterterm{A. The hash chain, precisely}{construction, properties, and honest limits}
\masterterminal{agent/trace.py — the construction}{%
step\_record = \{hypothesis, sql, statistic, verdict, ts\}\\
canonical\ \ \ = json.dumps(step\_record, sort\_keys=True)\\
step\_hash\ \ \ = sha256(prev\_hash + canonical).hexdigest()\\
chain.append(\{**step\_record, "hash": step\_hash\})\\
\# verify: replay the fold; any edit $\Rightarrow$ every later hash mismatches}
\vspace{2mm}
\SizeBody{\textbf{Guarantees:} integrity after sealing (tamper-evidence) and
\emph{reproducibility} — each step embeds its literal SQL and statistics, so a
verifier can re-run every query against the same dataset and reproduce the verdicts.\\
\textbf{Non-guarantees, stated before you ask:} execution attestation (mitigated by
re-runnability), author authentication (production answer: signing keys), and
storage-level rewrite of the entire chain plus root (production answer: anchor the
root hash externally). Tamper tests in CI attempt forgery on every commit.}

% ── C. SQL sandbox, attack-tree style ──
\masterterm{C. SQL sandbox defense map}{four layers, one scar}
\SizeBody{%
\textbf{L1 — engine:} DuckDB \texttt{enable\_external\_access=False} — filesystem
unreachable even if all other layers fail. The load-bearing wall.\\
\textbf{L2 — guard:} SELECT-only allowlist · comment rejection (smuggling vector) ·
catalog/schema-probe denial · table-function blocklist (\texttt{read\_csv} et al.).\\
\textbf{L3 — resources:} result-row cap, execution timeout, per-IP rate limit upstream.\\
\textbf{L4 — verification:} adversarial corpus in CI; grew by one entry the day it
caught a real file-read vulnerability in my own guard. The attack is now a permanent
regression test. Enumerating badness fails; removing capability works.}
